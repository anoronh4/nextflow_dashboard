# Split Dashboard into Source Files — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Move the dashboard's hand-written code out of the 4 MB single file into readable files under `src/` that run directly from `file://`, regenerate the single-file `pipeline_dashboard_offline.html` from them with a stdlib-only `build.py`, then make four behaviour-preserving readability cleanups.

**Architecture:** `src/index.html` loads classic `<script src>` / `<link rel=stylesheet>` files (no ES modules, which browsers refuse over `file://`), sharing globals exactly as today. `build.py` inlines them back, in document order, into the root bundle. The move is done by a one-off script that slices exact line ranges from the current file, so it is verbatim and checkable. Equivalence is proven by an in-browser snapshot harness that hashes every chart's Plotly data/layout, tables, metrics and exports, compared against a baseline from the original file.

**Tech Stack:** HTML/CSS/vanilla JS, Plotly 2.26.0 and Bootstrap 5.3.2 (vendored), Python 3 stdlib, Chrome via the chrome-devtools MCP tools for verification.

**Spec:** `docs/superpowers/specs/2026-11-13-split-into-source-files-design.md`

- [ ] **Step 2: Write the snapshot harness `.dev/snapshot.js`**

This is a single async arrow function, run with `evaluate_script` (`sourcePath`, `format: "function"`). It walks fixed UI states and returns `{key: value}`, where most values are short SHA-256 prefixes.

```js
async () => {
  const sleep = ms => new Promise(r => setTimeout(r, ms));
  const sha = async s => {
    const buf = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(s));
    return [...new Uint8Array(buf)].map(x => x.toString(16).padStart(2, '0')).join('').slice(0, 16);
  };
  // Plotly adds per-render uids and private keys; they are not part of what the chart shows.
  const clean = (k, v) => (k === 'uid' || (typeof k === 'string' && k.startsWith('_'))) ? undefined : v;

  // Swallow downloads, but keep the last one so the TSV can be hashed.
  let lastDownload = null;
  const origCOU = URL.createObjectURL.bind(URL);
  URL.createObjectURL = b => { lastDownload = b; return origCOU(b); };
  HTMLAnchorElement.prototype.click = function () { if (this.href.startsWith('data:')) lastDownload = this.href; };

  const out = {};
  const tabs = [...document.querySelectorAll('.tab-btn')].map(b => b.dataset.tab);
  const show = async tab => { switchTab(document.querySelector(`.tab-btn[data-tab="${tab}"]`)); await sleep(300); };

  async function capture(label, { exports = false } = {}) {
    for (const tab of tabs) {
      await show(tab);
      const pane = document.getElementById('pane-' + tab);
      const plots = [...pane.querySelectorAll('.js-plotly-plot')];
      for (const [i, gd] of plots.entries()) {
        out[`${label}/${tab}/plot${i}`] = await sha(JSON.stringify({ d: gd.data, l: gd.layout }, clean));
        out[`${label}/${tab}/plot${i}/modebar`] =
          [...gd.querySelectorAll('.modebar-btn')].map(b => b.getAttribute('data-title')).join('|');
        if (exports && gd.offsetParent)
          for (const th of ['dark', 'light']) out[`${label}/${tab}/plot${i}/png-${th}`] = await sha(await exportPlot(gd, th));
      }
      out[`${label}/${tab}/dom`] =
        await sha([...pane.querySelectorAll('table, .chart-empty, .metric-card')].map(e => e.outerHTML).join(''));
    }
    out[`${label}/metrics`] = document.getElementById('metric-grid').innerText;
  }

  const setRun = v => { document.getElementById('filter-run').value = v; updateAll(); };

  await capture('default', { exports: true });

  setRun(document.getElementById('filter-run').options[1].value);
  await capture('run1');
  setRun('');

  const keep = new Set(['BWAMEM2_MEM', 'GATK4_MARKDUPLICATES', 'STRELKA_SOMATIC']);
  allProcCounts().forEach((_, name) => { if (!keep.has(name)) state.excludedProcs.add(name); });
  renderProcList(); updateAll();
  await capture('procs3');
  state.excludedProcs.clear(); renderProcList(); updateAll();

  toggleStatus(document.getElementById('btn-CACHED'));
  await capture('nocached');
  toggleStatus(document.getElementById('btn-CACHED'));

  toggleAnonymize();
  await capture('anon');
  toggleAnonymize();

  setProcSort('alpha'); setProcColor('alpha');
  await capture('durations-alpha');
  setProcSort('median'); setProcColor('median');

  setWrittenMetric('wchar');
  await capture('wchar');
  setWrittenMetric('write_bytes');

  await show('typical');
  lastDownload = null;
  downloadTypicalTSV();
  await sleep(100);
  out['tsv'] = lastDownload == null ? 'NONE'
    : await sha(typeof lastDownload === 'string' ? lastDownload : await lastDownload.text());

  out['__keys'] = Object.keys(out).length;
  return out;
}
```

```python
#!/usr/bin/env python3
"""Bundle src/ into the single self-contained pipeline_dashboard_offline.html.

    python3 build.py           write pipeline_dashboard_offline.html
    python3 build.py --check   exit 1 if pipeline_dashboard_offline.html is out of date with src/

Standard library only. Every local <link rel="stylesheet" href="..."> and <script src="..."></script>
line in src/index.html is replaced by an inline <style>/<script> holding that file, in document order,
so the output works offline from a single file. Edit src/, never the generated file.
"""
import pathlib
import re
import sys

ROOT = pathlib.Path(__file__).resolve().parent
SRC = ROOT / 'src'
OUT = ROOT / 'pipeline_dashboard_offline.html'
BANNER = '<!-- GENERATED by build.py from src/ — edit src/, then run python3 build.py -->\n'
CSS_TAG = re.compile(r'^(\s*)<link rel="stylesheet" href="([^"]+)">\s*$')
JS_TAG = re.compile(r'^(\s*)<script src="([^"]+)"></script>\s*$')


def inline(rel, closing):
    """Contents of src/<rel>, refusing anything that would break offline use or the inline block."""
    if '://' in rel or rel.startswith(('/', '//')):
        sys.exit(f'build.py: {rel!r} is not a local file; the bundle must work offline')
    text = (SRC / rel).read_text(encoding='utf-8')
    if closing in text.lower():
        sys.exit(f'build.py: src/{rel} contains {closing!r}, which would end the inline block early')
    return text if text.endswith('\n') else text + '\n'


def build():
    parts = []
    for line in (SRC / 'index.html').read_text(encoding='utf-8').splitlines(keepends=True):
        if m := CSS_TAG.match(line):
            indent, rel = m.groups()
            parts.append(f'{indent}<!-- src/{rel} -->\n{indent}<style>\n{inline(rel, "</style")}</style>\n')
        elif m := JS_TAG.match(line):
            indent, rel = m.groups()
            parts.append(f'{indent}<!-- src/{rel} -->\n{indent}<script>\n{inline(rel, "</script")}</script>\n')
        else:
            parts.append(line)
    html = ''.join(parts)
    after_doctype = html.index('\n') + 1          # nothing may come before <!DOCTYPE html>
    return html[:after_doctype] + BANNER + html[after_doctype:]


def main():
    html = build()
    if '--check' in sys.argv[1:]:
        current = OUT.read_text(encoding='utf-8') if OUT.exists() else None
        if current != html:
            sys.exit('build.py --check: pipeline_dashboard_offline.html is out of date; run python3 build.py')
        print('build.py --check: up to date')
        return
    OUT.write_text(html, encoding='utf-8')
    print(f'wrote {OUT.name} ({len(html.encode()) / 1e6:.1f} MB)')


if __name__ == '__main__':
    main()
```

### Task 6: Shared event-sweep helpers

**Files:**
- Modify: `src/js/charts/concurrency.js`, `src/js/charts/queue.js`, `src/js/charts/typical.js`, `src/js/app.js`

There are two sweep shapes, and they must stay distinct to keep the output identical:
- **Stacked step series** (5 charts): groups events at the same timestamp and records the running total per sample before and after each timestamp.
- **Scalar peak** (metric card + 2 Typical Sample stats): applies events one at a time, in stable sort order, and keeps the max.

Memory charts sum **bytes** and convert to GB only when recording, so the helper takes a `record` transform. Converting up front would change float results.

- [ ] **Step 1: Add the helpers at the top of `src/js/charts/concurrency.js`**

Insert directly after the file's first banner block (the three `// ═══…` / `// CHART 1: …` lines):

```js
// ── Event sweeps ─────────────────────────────────────────────────────────
// Every concurrency curve is the same trick: each task adds +value when it starts and -value
// when it ends; sort those events by time and keep a running total.

// Stacked step series, one per sample. At each distinct timestamp the running totals are
// recorded twice, before and after applying that timestamp's events, so Plotly draws vertical
// steps instead of slopes. `record` transforms a total when it is stored (e.g. bytes → GB).
function stackedSteps(tasks, from, to, value, record = v => v) {
  const samples = [...new Set(tasks.map(t => t.sampleId))].sort();
  const evs = [];
  tasks.forEach(t => {
    const v = value(t);
    evs.push({ t: from(t).getTime(), d: +v, s: t.sampleId });
    evs.push({ t: to(t).getTime(),   d: -v, s: t.sampleId });
  });
  evs.sort((a, b) => a.t - b.t);

  const cur = {}, byS = {}, times = [];
  samples.forEach(s => { cur[s] = 0; byS[s] = []; });
  let i = 0;
  while (i < evs.length) {
    const t = evs[i].t;
    times.push(new Date(t));                                   // left edge of the step
    samples.forEach(s => byS[s].push(record(cur[s])));
    while (i < evs.length && evs[i].t === t) { cur[evs[i].s] += evs[i].d; i++; }
    times.push(new Date(t));                                   // right edge
    samples.forEach(s => byS[s].push(record(cur[s])));
  }
  return { times, byS, samples };
}

// Highest running total of `value` over tasks' start→complete windows (events applied one by
// one in time order). Used for the Peak CPUs card and per-sample peaks in Typical Sample.
function peakConcurrent(tasks, value) {
  const evs = tasks
    .flatMap(t => [[t.start.getTime(), +value(t)], [t.complete.getTime(), -value(t)]])
    .sort((a, b) => a[0] - b[0]);
  let cur = 0, peak = 0;
  evs.forEach(([, d]) => { cur += d; if (cur > peak) peak = cur; });
  return peak;
}

```
